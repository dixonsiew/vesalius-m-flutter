import 'dart:convert';

import 'package:flutter/cupertino.dart';
import 'package:flutter/material.dart';
import 'package:modal_progress_hud_nsn/modal_progress_hud_nsn.dart';
import 'package:vesalius_m_flutter/components/app-shared.dart';
import 'package:vesalius_m_flutter/constants.dart';
import 'package:vesalius_m_flutter/models/data-manager.dart';
import 'package:vesalius_m_flutter/models/doctor-data.dart';
import 'package:vesalius_m_flutter/services/data-service.dart';
import 'package:vesalius_m_flutter/helpers.dart';

class DoctorList extends StatefulWidget {
  
  static const String routeName = 'DoctorList';

  final String specialtyCode;

  DoctorList({
    @required this.specialtyCode,
  });

  @override
  _DoctorListState createState() => _DoctorListState();
}

class _DoctorListState extends State<DoctorList> {

  List<DoctorDetails> list = [];
  List<DoctorDetails> _list = [];
  bool isLoading = false;

  final searchController = TextEditingController();
  final GlobalKey<RefreshIndicatorState> refreshIndicatorKey = GlobalKey<RefreshIndicatorState>();

  @override
  void initState() {
    super.initState();
    load();
  }

  void load() async {
    try {
      setState(() {
        isLoading = true;
      });
      var branchDetails = DataManager.branchDetails;
      var lx = await getVesaliusDoctorData(branchDetails.branch.branchId);
      _list = lx ?? [];
      var lm = filterDoctorListBasedOnSpecialtyCode();
      setState(() {
        list = lm;
        _list = lx ?? [];
        isLoading = false;
      });

      if (_list.isEmpty) {
        showCustomDialog('Loading Doctor Data Failed', 'Please Try again Later', 'Dismiss', context);
      }
    }

    catch (error) {
      setState(() {
        isLoading = false;
      });
      showCustomDialog('Failed', 'Unable to retrieve doctor information at the moment. Please check your internet connection or try again later.', 'Dismiss', context);
    }
  }

  Future<void> onRefresh() async {
    load();
  }

  List<DoctorDetails> filterDoctorListBasedOnSpecialtyCode() {
    if (widget.specialtyCode == null || widget.specialtyCode == '') {
      return _list;
    }

    var la = _list.where((x) => x.specialtyList.any((y) => y.code == widget.specialtyCode)).toList();
    if (la.isEmpty) {
      showCustomDialog('Failed', 'Sorry, no doctor found based on the selected specialty. Please reset and search again.', 'Dismiss', context);
    }

    return la;
  }

  void filterDoctor(String s) {
    var lm = filterDoctorListBasedOnSpecialtyCode();
    // ignore: null_aware_in_condition
    if (s?.isEmpty) {
      setState(() {
        list = lm;
      });
    }

    else {
      String r = s.toLowerCase();
      var q = lm.where((o) {
        var name = o.name;
        String title = name.title;
        bool btitle = title?.toLowerCase()?.contains(r);
        String firstName = name.firstName;
        bool bfirstName = firstName?.toLowerCase()?.contains(r);
        String lastName = name.lastName;
        bool blastName = lastName?.toLowerCase()?.contains(r);
        String fullname = '$firstName $lastName';
        bool bfullname = fullname.toLowerCase().contains(r);
        String fullnameWithTitle = '$title $firstName $lastName';
        bool bfullnameWithTitle = fullnameWithTitle.toLowerCase().contains(r);
        List<Specialty> specialtyList = o.specialtyList;
        bool bspecialtyList = false;
        if (specialtyList != null) {
          var ls = specialtyList.map((e) {
            String x = e.description;
            return x;
          });
          String x = ls.join(', ');
          bspecialtyList = x.toLowerCase().contains(r);
        }
        
        List<String> qualification = o.qualification;
        bool bqualification = false;
        if (qualification != null) {
          bqualification = qualification.any((e) {
            return e.toLowerCase().contains(r);
          });
        }

        return btitle || bfirstName || blastName || bfullname || bfullnameWithTitle || bspecialtyList || bqualification;
      });
      setState(() {
        list = q.toList();
      });
    }
  }

  ImageProvider<Object> getDoctorImage(DoctorDetails o) {
    var image = o.image;
    String imageData = image.base64ImageData;
    ImageProvider<Object> im = AssetImage('images/imgs/no_image.png');
    if (imageData != null && imageData != '') {
      im = MemoryImage(
        base64Decode(imageData.trim()),
      );
    }

    return im;
  }

  Widget buildContent(DoctorDetails o) {
    if (isLoading) {
      return Container();
    }

    Name name = o.name;

    return InkWell(
      onTap: () {
        Navigator.pop(context, o);
      },
      child: Padding(
        padding: EdgeInsets.all(20.0),
        child: Row(
          children: [
            Container(
              margin: EdgeInsets.only(right: 40.0),
              width: 50.0,
              height: 50.0,
              decoration: BoxDecoration(
                shape: BoxShape.circle,
                image: DecorationImage(
                  image: getDoctorImage(o),
                  fit: BoxFit.cover,
                ),
              ),
            ),
            Flexible(
              child: Text(
                '${name.title} ${name.firstName}'.trim(),
                style: TextStyle(
                  color: Colors.black,
                  fontSize: 18.0,
                ),
              ),
            ),
          ],
        ),
      ),
    );
  }

  Widget buildSearch() {
    return Padding(
      padding: EdgeInsets.only(left: 15.0, right: 15.0, top: 15.0),
      child: TextField(
        controller: searchController,
        cursorColor: Color(0xFF999494),
        decoration: InputDecoration(
          isDense: true,
          contentPadding: EdgeInsets.symmetric(horizontal: 15.0),
          hintText: 'Search',
          prefixIcon: Icon(
            Icons.search,
            color: Color(0xFF999494),
          ),
          enabledBorder: OutlineInputBorder(
            borderRadius: BorderRadius.all(Radius.circular(5.0)),
            borderSide: BorderSide(color: Color(0xFF999494)),
          ),
          focusedBorder: OutlineInputBorder(
            borderRadius: BorderRadius.all(Radius.circular(5.0)),
            borderSide: BorderSide(color: Color(0xFF999494)),
          ),
        ),
        onChanged: (String s) {
          filterDoctor(s);
        },
      ),
    );
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        brightness: Brightness.dark,
        backgroundColor: kAppointmentBgColor,
        toolbarHeight: kAppToolbarHeight,
        automaticallyImplyLeading: false,
        centerTitle: true,
        title: Text(
          'Select Doctor',
          style: TextStyle(
            color: Colors.white,
            fontSize: 18.0,
            fontWeight: FontWeight.bold,
          ),
        ),
        actions: [
          IconButton(
            icon: Icon(
              Icons.close,
              color: Colors.white,
            ),
            onPressed: () {
              Navigator.pop(context);
            }
          ),
        ],
      ),
      backgroundColor: Color(0xFFF5F5F5),
      body: ModalProgressHUD(
        inAsyncCall: isLoading,
        progressIndicator: AppScalingText('Loading...'),
        child: SafeArea(
          child: RefreshIndicator(
            key: refreshIndicatorKey,
            onRefresh: onRefresh,
            color: kPrimaryColor,
            child: Scrollbar(
              child: ListView.separated(
                shrinkWrap: true,
                itemCount: list.length + 1,
                itemBuilder: (context, i) {
                  if (i == 0) {
                    return buildSearch();
                  }

                  return buildContent(list[i - 1]);
                },
                separatorBuilder: (context, i) {
                  if (i > 0 && !isLoading) {
                    return Divider(
                      height: 1.0,
                      thickness: 1.0,
                      color: Color.fromARGB(225, 226, 226, 226),
                    );
                  }
                  
                  return Container();
                },
              ),
            ),
          ),
        ),
      ),
    );
  }
}
